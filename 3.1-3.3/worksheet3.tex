    \part{Determinants}
    \section{\centering Determinants and 
    Transformations (3.1) -- (3.3)}
    \begin{mideea}{0.9}
        \begin{enumerate}
            \item \(\det A = 0 \iff A \ \text{is not invertible}\).
            \item Scaling any row by a constant
            scales the determinant by the same
            amount.
            \item \(\det(AB) = \det A \cdot \det B\)
            \item \(\det A = \det A^T\)
        \end{enumerate}
    \end{mideea}

    \begin{newactivity}
        Use the statements in
        \textbf{Main Idea \#1} to answer the 
        following questions.

        \begin{enumerate}
            \item If \(
                \begin{vmatrix}
                    a & b & c \\
                    d & e & f \\
                    g & h & i
                \end{vmatrix} = 6
                \), what's \(
                \begin{vmatrix}
                    g &  2h & i \\
                    a &  2b & c \\
                    3d & 6e & 3f \\
                \end{vmatrix}
                \)?
            \item What's 
                \(\begin{vmatrix}
                    a & b & 2a + b \\
                    c & d & 2c + d \\
                    e & f & 2e + f
            \end{vmatrix}\)?
            \item If \(A = LU\), can \(U\) be 
                  invertible while \(A\) isn't?
                  (\textit{Hint: Reason with 
                  determinants, and remember 
                  that \(L\) is lower 
                  triangular.})
            \item What's the area of the 
                  parallelogram whose vertices 
                  are \((0, 0), (1, 2), (3, 1), (4, 3)\)? What's the area after
                  applying a mapping \(\vec{x} \mapsto \begin{bmatrix}
                      3 & 0 \\ 0 & 1
                  \end{bmatrix}\vec{x}\) on
                  the vertices?
        \end{enumerate}
    \end{newactivity}
