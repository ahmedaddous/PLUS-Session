\section{\centering Construction Questions}
	\begin{newquestion}
		For each question, provide an example that meets the specifications.
		\begin{enumerate}
			\item Find a \(2 \times 3\) matrix \(A\) with \(\nullspace A^T = \vspan \left\{\!\! \begin{pmatrix}1 \\ 2\end{pmatrix}\!\!\right\}\)
			\item Find a matrix that is diagonalizable but not invertible.
			\item Find a matrix \(A\) with \(\nullspace A = \vspan\left\{\!\!\begin{pmatrix}1 \\ 2 \\ 0\end{pmatrix}, \begin{pmatrix}1 \\ 0 \\ 1\end{pmatrix}\!\!\right\}\) and \(\rowspace A = \vspan\left\{\!\!\begin{pmatrix}-2 \\ 1 \\ 2\end{pmatrix}\!\!\right\}\) (Note that this vector was not chosen by coincidence.)
			\item Construct a \(3 \times 3\) matrix with 2 distinct eigenvalues, \(\lambda_1 = 2\) and \(\lambda_2 = \lambda_3 = 1\). The second distinct
			eigenvalue must have a geometric multiplicity of 1. Let the matrix \emph{not} be a diagonal matrix. 
		\end{enumerate}
	\end{newquestion}

			\end{enumerate}
	\end{enumerate}
